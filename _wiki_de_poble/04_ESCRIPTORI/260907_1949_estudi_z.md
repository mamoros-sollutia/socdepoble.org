## 1. Veredicte

El front-end ja té l'arquitectura d'Identitats preparada (`/jo` + `/e/:slug`) i la infraestructura d'auth parcialment presents (callback OAuth, relay, capa Supabase). **La Fase 3 no és una construcció des de zero: és tancar el circuit backend.** Es necessiten 5 taules, 3 funcions RPC, 1 trigger, polítiques RLS completes i l'ampliació de dos contexts existents.

**SDP-LOCK: no s'activa.** La migració proposada és purament additiva (taules noves, cap `DROP`, cap `TRUNCATE`, cap UPDATE sobre dades existents). El trigger sobre `auth.users` només fa `INSERT` a `public.perfils`.

---

## 2. Decisions de disseny (amb raó)

| Decisió | Opció triada | Per què |
|---|---|---|
| Perfil d'usuari | Taula `public.perfils` 1:1 amb `auth.users` | `raw_user_meta_data` no és fiable ni consultable amb RLS. Mai hi posem rols. |
| Rols globals | Taula `rols_globals` + funció `es_superadmin()` | Auditable, consultable des de RLS, i evita tocar `auth.users`. |
| Relació usuari–entitat | N:M amb `entitat_membres` i rol per filera | Un usuari pot gestionar diverses entitats; una entitats, diversos membres. Encaixa amb `/e/:slug` múltiples. |
| Propietat d'entitats | Model híbrid: **creació lliure** + **reclamació verificada** per al catàleg importat | Les entitats creades per usuaris són seves d'immediat; les importades del catàleg del poble passen per aprovació del Superadmin. |
| Operacions sensibles | RPC `SECURITY DEFINER` amb validació interna | Evita polítiques RLS d'INSERT fràgils a `entitat_membres` (ningú es pot autoproclamar membre). |
| Estat de sessió React | Ampliació del `SessionContext` existent | Zero dependències noves. |

### 3.1 Taules

```sql
-- ============================================================
-- FASE 3 · Identitats, entitats, rols i reclamacions
-- Additiu. No toca dades existents. SDP-LOCK: no aplicable.
-- ============================================================

-- 1. PERFILS (1:1 amb auth.users, dades PÚBLIQUES del perfil)
create table public.perfils (
  id              uuid primary key references auth.users(id) on delete cascade,
  usuari          text not null,                    -- nom públic visible
  slug            text,                             -- opcional, futur /u/:slug
  presentacio     text,
  avatar_url      text,
  creat_a         timestamptz not null default now(),
  actualitzat_a   timestamptz not null default now()
);
create unique index perfils_slug_unic on public.perfils (slug) where slug is not null;
create index perfils_usuari_idx on public.perfils (usuari);

-- 2. ENTITATS
create table public.entitats (
  id          uuid primary key default gen_random_uuid(),
  slug        text not null unique,
  nom         text not null,
  descripcio  text,
  tipus       text not null check (tipus in ('empresa','ajuntament','associacio','altre')),
  logo_url    text,
  estat       text not null default 'pendent'
              check (estat in ('pendent','activa','suspesa')),
  creat_a     timestamptz not null default now(),
  actualitzat_a timestamptz not null default now()
);

-- 3. MEMBRES (relació usuari-entitat amb rol)
create table public.entitat_membres (
  id          uuid primary key default gen_random_uuid(),
  entitat_id  uuid not null references public.entitats(id) on delete cascade,
  perfil_id   uuid not null references public.perfils(id) on delete cascade,
  rol         text not null default 'editor'
              check (rol in ('propietari','admin','editor','lector')),
  creat_a     timestamptz not null default now(),
  unique (entitat_id, perfil_id)
);
create index entitat_membres_perfil_idx on public.entitat_membres (perfil_id);

-- 4. RECLAMACIONS de propietat (només per a entitats del catàleg)
create table public.entitat_reclamacions (
  id          uuid primary key default gen_random_uuid(),
  entitat_id  uuid not null references public.entitats(id) on delete cascade,
  perfil_id   uuid not null references public.perfils(id) on delete cascade,
  estat       text not null default 'pendent'
              check (estat in ('pendent','aprovada','rebutjada','retirada')),
  missatge    text,           -- Raó de la reclamació (lliure)
  contacte    text,           -- Dada de verificació (correu oficial, telèfon)
  creat_a     timestamptz not null default now(),
  resolt_a    timestamptz
);
create index reclamacions_estat_idx on public.entitat_reclamacions (estat);

-- 5. ROLS GLOBALS (superadministradors)
create table public.rols_globals (
  perfil_id   uuid primary key references public.perfils(id) on delete cascade,
  rol         text not null default 'superadmin',
  creat_a     timestamptz not null default now()
);
```

### 3.2 Funcions

```sql
-- Funció de verificació de rol global (usable des de RLS)
create or replace function public.es_superadmin(uid uuid)
returns boolean
language sql stable security definer set search_path = public
as $$
  select exists (select 1 from public.rols_globals where perfil_id = uid);
$$;

-- Creació automàtica de perfil en registrar-se (patró canònic Supabase)
create or replace function public.crear_perfil_darrere_usuari()
returns trigger
language plpgsql security definer set search_path = public
as $$
begin
  insert into public.perfils (id, usuari)
  values (new.id, 'vei_' || substr(new.id::text, 1, 8));
  return new;
end;
$$;

create trigger on_auth_user_created
  after insert on auth.users
  for each row execute function public.crear_perfil_darrere_usuari();

-- RPC: crear entitat pròpia (transacció segura: entitat + propietariat)
create or replace function public.crear_entitat(
  p_nom text, p_slug text, p_tipus text, p_descripcio text default null
)
returns uuid
language plpgsql security definer set search_path = public
as $$
declare v_id uuid;
begin
  if auth.uid() is null then
    raise exception 'No autenticat';
  end if;
  insert into public.entitats (nom, slug, tipus, descripcio, estat)
  values (p_nom, p_slug, p_tipus, p_descripcio, 'activa')
  returning id into v_id;

  insert into public.entitat_membres (entitat_id, perfil_id, rol)
  values (v_id, auth.uid(), 'propietari');

  return v_id;
end;
$$;

-- RPC: resoldre una reclamació (només Superadmin)
create or replace function public.resoldre_reclamacio(
  p_reclamacio uuid, p_estat text
)
returns void
language plpgsql security definer set search_path = public
as $$
declare v_entitat uuid; v_perfil uuid;
begin
  if not public.es_superadmin(auth.uid()) then
    raise exception 'No autoritzat';
  end if;
  if p_estat not in ('aprovada','rebutjada') then
    raise exception 'Estat no vàlid';
  end if;

  select entitat_id, perfil_id into v_entitat, v_perfil
  from public.entitat_reclamacions
  where id = p_reclamacio and estat = 'pendent';

  if v_entitat is null then
    raise exception 'Reclamació inexistent o ja resolta';
  end if;

  if p_estat = 'aprovada' then
    insert into public.entitat_membres (entitat_id, perfil_id, rol)
    values (v_entitat, v_perfil, 'propietari')
    on conflict (entitat_id, perfil_id) do update set rol = 'propietari';

    update public.entitats set estat = 'activa', actualitzat_a = now()
    where id = v_entitat;
  end if;

  update public.entitat_reclamacions
  set estat = p_estat, resolt_a = now()
  where id = p_reclamacio;
end;
$$;

-- updated_at automàtic
create or replace function public.toca_actualitzat_a()
returns trigger language plpgsql as $$
begin
  new.actualitzat_a = now();
  return new;
end;
$$;
create trigger perfils_toca before update on public.perfils
  for each row execute function public.toca_actualitzat_a();
create trigger entitats_toca before update on public.entitats
  for each row execute function public.toca_actualitzat_a();
```

### 3.3 RLS (policies)

```sql
alter table public.perfils            enable row level security;
alter table public.entitats           enable row level security;
alter table public.entitat_membres    enable row level security;
alter table public.entitat_reclamacions enable row level security;
alter table public.rols_globals       enable row level security;

-- PERFILS: lectura pública dels camps públics; edició només pròpia.
create policy perfils_select on public.perfils
  for select using (true);
create policy perfils_update on public.perfils
  for update using (auth.uid() = id);
-- INSERT: no cal policy. Ho fa el trigger (SECURITY DEFINER).

-- ENTITATS: públic només les actives; Superadmin ho veu tot.
create policy entitats_select on public.entitats
  for select using (
    estat = 'activa' or public.es_superadmin(auth.uid())
  );
-- INSERT/UPDATE/DELETE: via RPC o Superadmin.
create policy entitats_admin on public.entitats
  for all using (public.es_superadmin(auth.uid()));

-- MEMBRES: cadascú veu els seus; admins de l'entitat veuen tots els membres.
create policy membres_select on public.entitat_membres
  for select using (
    perfil_id = auth.uid()
    or exists (
      select 1 from public.entitat_membres m
      where m.entitat_id = entitat_id
        and m.perfil_id = auth.uid()
        and m.rol in ('propietari','admin')
    )
    or public.es_superadmin(auth.uid())
  );
-- INSERT/DELETE: via RPC o Superadmin. Cap policy d'insert directe.

-- RECLAMACIONS: cadascú veu les seves; Superadmin les veu totes.
create policy reclamacions_select on public.entitat_reclamacions
  for select using (
    perfil_id = auth.uid() or public.es_superadmin(auth.uid())
  );
create policy reclamacions_insert on public.entitat_reclamacions
  for insert with check (
    perfil_id = auth.uid()
    and exists (select 1 from public.entitats e
                where e.id = entitat_id and e.estat = 'pendent')
  );
create policy reclamacions_retirar on public.entitat_reclamacions
  for delete using (perfil_id = auth.uid() and estat = 'pendent');

-- ROLS GLOBALS: visible pel propietari i pel Superadmin.
create policy rols_select on public.rols_globals
  for select using (perfil_id = auth.uid() or public.es_superadmin(auth.uid()));
-- Gestió: només des de consola SQL amb rol service_role. Cap policy d'escritura.
```

  const [perfil, setPerfil] = useState(null);
  const [entitats, setEntitats] = useState([]);   // entitats que gestiona

      setCarregant(false);

    const { data: sub } = supabase.auth.onAuthStateChange((_e, s) => {
      setSession(s);
      if (!s) { setPerfil(null); setEntitats([]); }
    });
    return () => sub.subscription.unsubscribe();
  }, []);

  // Càrrega d'identitat quan hi ha sessió

        supabase.from('perfils').select('*').eq('id', uid).single(),

      setPerfil(p.data);
      setEntitats(m.data ?? []);
      setOnboardingPendent(Boolean(p.data) && !p.data?.actualitzat_manual); // ⚠️ flag
    })();

## 8. Base legal i privacitat (LLEI_05_Privacitat)

- **Minimització:** el registre demana correu (GoTrue), nom públic i contrasenya. Res més. Presentació, avatar i contacte de reclamació: opcionals, posteriors, explicats.

- **Reclamacions:** la dada `contacte` és dada personal amb finalitat de verificació: emmagatzemar-la només mentre la reclamació estiga `pendent` (neteja en tancar — la RPC ja marca `resolt_a`; purgar `contacte` en resoldre: afegir `set contacte = null` dins `resoldre_reclamacio`).
